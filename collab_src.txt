import bcrypt from 'bcryptjs';
import { z } from 'zod';
import { prisma } from '@/lib/prisma';
import { startSession } from '@/lib/auth';
import { fail, limited, parse } from '@/lib/api';

const S = z.object({ email: z.string().trim().toLowerCase().email(), password: z.string().min(1).max(128) });
export async function POST(req: Request) {
  if (limited(req, 'login')) return fail('Too many attempts, try again later', 429);
  const d = await parse(req, S);
  if (!d) return fail('Enter a valid email and password');
  const u = await prisma.user.findUnique({ where: { email: d.email } });
  if (!u || !(await bcrypt.compare(d.password, u.passwordHash))) return fail('Invalid credentials', 401);
  await startSession(u.id);
  return NextResponse.json({ ok: true });
}

import { NextResponse } from 'next/server';
import { z } from 'zod';
import { prisma } from '@/lib/prisma';
import { fail, gate, parse } from '@/lib/api';

type Ctx = { params: { pageId: string } };
const S = z.object({ email: z.string().trim().toLowerCase().email(), permission: z.enum(['VIEW', 'COMMENT', 'EDIT']) });

export async function GET(_: Request, { params }: Ctx) {
  const g = await gate(params.pageId, 'VIEW', true);
  if ('res' in g) return g.res;
  const [page, rows] = await Promise.all([
    prisma.page.findUniqueOrThrow({ where: { id: params.pageId }, select: { workspace: { select: { owner: { select: { name: true, email: true } } } } } }),
    prisma.pagePermission.findMany({ where: { pageId: params.pageId }, include: { user: { select: { name: true, email: true } } } }),
  ]);
  return NextResponse.json({
    owner: page.workspace.owner,
    entries: rows.map((r) => ({ userId: r.userId, name: r.user.name, email: r.user.email, permission: r.permission })),
  });
}
export async function PUT(req: Request, { params }: Ctx) {
  const g = await gate(params.pageId, 'VIEW', true);
  if ('res' in g) return g.res;
  const d = await parse(req, S);
  if (!d) return fail('Enter a valid email and permission');
  const t = await prisma.user.findUnique({ where: { email: d.email }, select: { id: true } });
  if (!t) return fail('No registered user with that email', 404);
  const ws = await prisma.page.findUniqueOrThrow({ where: { id: params.pageId }, select: { workspace: { select: { ownerId: true } } } });
  if (ws.workspace.ownerId === t.id) return fail('Workspace owner always has full access');
  await prisma.pagePermission.upsert({
    where: { pageId_userId: { pageId: params.pageId, userId: t.id } },
    update: { permission: d.permission },
    create: { pageId: params.pageId, userId: t.id, permission: d.permission },
  });
  return NextResponse.json({ ok: true });
}
export async function DELETE(req: Request, { params }: Ctx) {
  const g = await gate(params.pageId, 'VIEW', true);
  if ('res' in g) return g.res;
  const userId = new URL(req.url).searchParams.get('userId');
  if (!userId) return fail('userId required');
  await prisma.pagePermission.deleteMany({ where: { pageId: params.pageId, userId } });
  return NextResponse.json({ ok: true });
}

export async function POST(req: Request, { params }: Ctx) {
  const g = await gate(params.pageId, 'EDIT');
  if ('res' in g) return g.res;
  const d = await parse(req, z.object({ label: z.string().trim().max(100).optional() }));
  if (!d) return fail('Invalid body');
  await new Promise((r) => setTimeout(r, 700)); // let the collab server's 500ms persistence batch flush
  const rows = await prisma.documentUpdate.findMany({ where: { pageId: params.pageId }, orderBy: { id: 'asc' }, select: { data: true } });
  const doc = new Y.Doc();

  const v = await prisma.documentVersion.create({
    data: { pageId: params.pageId, createdById: g.user.id, label: d.label || null, data: Buffer.from(Y.encodeStateAsUpdate(doc)) },
  });
  return NextResponse.json({ id: v.id });

    update: {},

  });

        title={page.title}

/**
 * Yjs collaboration server: ws + y-protocols, Postgres persistence, Redis pub/sub fan-out.
 * Run N instances (different COLLABORATION_PORT) against the same Postgres + Redis.
 */
import http from 'http';
import { randomUUID } from 'crypto';
import { WebSocketServer, WebSocket } from 'ws';
import Redis from 'ioredis';
import * as Y from 'yjs';
import * as sync from 'y-protocols/sync';
import * as awarenessProtocol from 'y-protocols/awareness';
import * as encoding from 'lib0/encoding';
import * as decoding from 'lib0/decoding';
import { prisma } from '../lib/prisma';
import { COOKIE, verifySession } from '../lib/session';
import { getPerm, type Perm } from '../lib/permissions';

const PORT = Number(process.env.COLLABORATION_PORT ?? 1234);
const APP_ORIGIN = new URL(process.env.NEXT_PUBLIC_APP_URL ?? 'http://localhost:3000').origin;
const INSTANCE = randomUUID(); // 36 chars; used to ignore our own Redis messages
const MSG_SYNC = 0;
const MSG_AWARENESS = 1;
const FLUSH_MS = 500; // batch keystrokes into one DB row
const COMPACT_AFTER = 50; // merge update rows after N writes

const pub = new Redis(process.env.REDIS_URL!);
const sub = new Redis(process.env.REDIS_URL!);

type Room = {
  id: string;
  doc: Y.Doc;
  awareness: awarenessProtocol.Awareness;
  conns: Map<WebSocket, Set<number>>; // conn -> awareness client ids it controls
  buf: Uint8Array[];
  timer?: NodeJS.Timeout;
  writes: number;
  ready: Promise<void>;
};
const rooms = new Map<string, Room>();
const chan = (id: string) => `collab:${id}`;
const frame = (kind: number, payload: Uint8Array) => Buffer.concat([Buffer.from([kind]), Buffer.from(INSTANCE), Buffer.from(payload)]);
const send = (c: WebSocket, m: Uint8Array) => { if (c.readyState === WebSocket.OPEN) c.send(m, (e) => e && c.close()); };

/* ---------- persistence ---------- */
async function compact(pageId: string) {
  const rows = await prisma.documentUpdate.findMany({ where: { pageId }, orderBy: { id: 'asc' }, select: { id: true, data: true } });
  if (rows.length < 2) return;
  const merged = Y.mergeUpdates(rows.map((r) => r.data));
  await prisma.$transaction([
    prisma.documentUpdate.deleteMany({ where: { id: { in: rows.map((r) => r.id) } } }), // only rows we read; concurrent inserts survive
    prisma.documentUpdate.create({ data: { pageId, data: Buffer.from(merged) } }),
  ]);
}

async function flush(room: Room) {
  room.timer = undefined;
  if (!room.buf.length) return;
  const merged = Y.mergeUpdates(room.buf.splice(0));
  try {
    await prisma.documentUpdate.create({ data: { pageId: room.id, data: Buffer.from(merged) } });
  } catch (e) {
    if ((e as { code?: string }).code === 'P2003') { room.conns.forEach((_, c) => c.close(4404)); return; } // page deleted
    console.error('persist failed, will retry', e);
    room.buf.unshift(merged); // never drop updates
    room.timer = setTimeout(() => flush(room), 2000);
    return;
  }
  prisma.page.update({ where: { id: room.id }, data: { updatedAt: new Date() } }).catch(() => {});
  if (++room.writes >= COMPACT_AFTER) {
    room.writes = 1;
    compact(room.id).catch((e) => console.error('compact failed', e));
  }
}

/* ---------- room lifecycle ---------- */
function onDocUpdate(room: Room, update: Uint8Array, origin: unknown) {
  const e = encoding.createEncoder();
  encoding.writeVarUint(e, MSG_SYNC);
  sync.writeUpdate(e, update);
  const m = encoding.toUint8Array(e);
  room.conns.forEach((_, c) => { if (c !== origin) send(c, m); });
  if (origin === 'redis' || origin === 'db') return; // already persisted/published by the originating instance
  pub.publish(chan(room.id), frame(0, update)).catch(console.error);
  room.buf.push(update);
  if (!room.timer) room.timer = setTimeout(() => flush(room), FLUSH_MS);
}

function onAwareness(room: Room, ch: { added: number[]; updated: number[]; removed: number[] }, origin: unknown) {
  if (origin instanceof WebSocket) {
    const ids = room.conns.get(origin);
    if (ids) { ch.added.forEach((i) => ids.add(i)); ch.removed.forEach((i) => ids.delete(i)); }
  }
  const upd = awarenessProtocol.encodeAwarenessUpdate(room.awareness, [...ch.added, ...ch.updated, ...ch.removed]);
  const e = encoding.createEncoder();
  encoding.writeVarUint(e, MSG_AWARENESS);
  encoding.writeVarUint8Array(e, upd);
  const m = encoding.toUint8Array(e);
  room.conns.forEach((_, c) => send(c, m));
  if (origin !== 'redis') pub.publish(chan(room.id), frame(1, upd)).catch(console.error);
}

function getRoom(id: string): Room {
  const existing = rooms.get(id);
  if (existing) return existing;
  const doc = new Y.Doc();
  const awareness = new awarenessProtocol.Awareness(doc);
  awareness.setLocalState(null);
  const room: Room = { id, doc, awareness, conns: new Map(), buf: [], writes: 0, ready: Promise.resolve() };
  rooms.set(id, room);
  room.ready = (async () => {
    await sub.subscribe(chan(id)); // subscribe before loading so no cross-instance update is missed
    const rows = await prisma.documentUpdate.findMany({ where: { pageId: id }, orderBy: { id: 'asc' }, select: { data: true } });
    doc.transact(() => rows.forEach((r) => Y.applyUpdate(doc, r.data)), 'db');
    room.writes = rows.length;
    doc.on('update', (u: Uint8Array, o: unknown) => onDocUpdate(room, u, o));
    awareness.on('update', (c: { added: number[]; updated: number[]; removed: number[] }, o: unknown) => onAwareness(room, c, o));
  })().catch((e) => {
    console.error('room load failed', id, e);
    rooms.delete(id);
    room.conns.forEach((_, c) => c.close(1011));
  });
  return room;
}

async function leave(room: Room, ws: WebSocket) {
  const ids = room.conns.get(ws);
  room.conns.delete(ws);
  if (ids?.size) awarenessProtocol.removeAwarenessStates(room.awareness, [...ids], null);
  if (room.conns.size) return;
  await room.ready;
  if (room.timer) clearTimeout(room.timer);
  await flush(room);
  if (room.conns.size || rooms.get(room.id) !== room) return; // someone joined meanwhile
  rooms.delete(room.id);
  await sub.unsubscribe(chan(room.id)).catch(() => {});
  await compact(room.id).catch((e) => console.error('compact failed', e));
  room.awareness.destroy();
  room.doc.destroy();
}

/* ---------- protocol ---------- */
function onMessage(room: Room, conn: WebSocket, perm: Perm, data: Uint8Array) {
  const dec = decoding.createDecoder(data);
  const type = decoding.readVarUint(dec);
  if (type === MSG_SYNC) {
    // Read-only users may only request state (step1). Their step2/update messages are dropped.
    if (decoding.peekVarUint(dec) !== sync.messageYjsSyncStep1 && perm !== 'EDIT') return;
    const e = encoding.createEncoder();
    encoding.writeVarUint(e, MSG_SYNC);
    sync.readSyncMessage(dec, e, room.doc, conn);
    if (encoding.length(e) > 1) send(conn, encoding.toUint8Array(e));
  } else if (type === MSG_AWARENESS) {
    awarenessProtocol.applyAwarenessUpdate(room.awareness, decoding.readVarUint8Array(dec), conn);
  }
}

sub.on('messageBuffer', (ch: Buffer, msg: Buffer) => {
  const room = rooms.get(ch.toString().slice('collab:'.length));
  if (!room || msg.subarray(1, 37).toString() === INSTANCE) return;
  const payload = new Uint8Array(msg.subarray(37));
  try {
    if (msg[0] === 0) Y.applyUpdate(room.doc, payload, 'redis');
    else awarenessProtocol.applyAwarenessUpdate(room.awareness, payload, 'redis');
  } catch (e) { console.error('bad redis message', e); }
});

/* ---------- server ---------- */
const server = http.createServer((_q, res) => { res.writeHead(200); res.end('ok'); });
const wss = new WebSocketServer({ noServer: true, maxPayload: 5 * 1024 * 1024 });
type Ctx = { uid: string; pageId: string; perm: Perm };

server.on('upgrade', async (req, socket, head) => {
  const reject = (code: number, text: string) => { socket.write(`HTTP/1.1 ${code} ${text}\r\nConnection: close\r\n\r\n`); socket.destroy(); };
  try {
    if (req.headers.origin !== APP_ORIGIN) return reject(403, 'Forbidden');
    const pageId = decodeURIComponent((req.url ?? '').split('?')[0].slice(1));
    if (!pageId || pageId.length > 64) return reject(400, 'Bad Request');
    const token = req.headers.cookie?.split(';').map((s) => s.trim().split('=')).find(([k]) => k === COOKIE)?.[1];
    const uid = await verifySession(token);
    if (!uid) return reject(401, 'Unauthorized');
    const perm = await getPerm(uid, pageId);
    if (!perm) return reject(403, 'Forbidden');
    wss.handleUpgrade(req, socket, head, (ws) => wss.emit('connection', ws, req, { uid, pageId, perm } satisfies Ctx));
  } catch (e) {
    console.error(e);
    reject(500, 'Internal Server Error');
  }
});

wss.on('connection', (ws: WebSocket, _req: http.IncomingMessage, ctx: Ctx) => {
  const room = getRoom(ctx.pageId);
  room.conns.set(ws, new Set());
  const ready = room.ready;

  ws.on('message', async (data: Buffer) => {
    await ready;
    if (ws.readyState !== WebSocket.OPEN) return;
    try { onMessage(room, ws, ctx.perm, new Uint8Array(data)); } catch (e) { console.error(e); ws.close(1003); }
  });

  let alive = true;
  ws.on('pong', () => { alive = true; });
  const ping = setInterval(() => { if (!alive) return ws.terminate(); alive = false; ws.ping(); }, 30_000);
  // Re-check permission so revocation/downgrade applies to live connections.
  const recheck = setInterval(async () => {
    const p = await getPerm(ctx.uid, ctx.pageId).catch(() => ctx.perm);
    if (!p) ws.close(4403); else ctx.perm = p;
  }, 15_000);
  ws.on('close', () => { clearInterval(ping); clearInterval(recheck); void leave(room, ws); });

  ready.then(() => {
    if (ws.readyState !== WebSocket.OPEN) return;
    const e = encoding.createEncoder();
    encoding.writeVarUint(e, MSG_SYNC);
    sync.writeSyncStep1(e, room.doc);
    send(ws, encoding.toUint8Array(e));
    const states = room.awareness.getStates();
    if (states.size) {
      const a = encoding.createEncoder();
      encoding.writeVarUint(a, MSG_AWARENESS);
      encoding.writeVarUint8Array(a, awarenessProtocol.encodeAwarenessUpdate(room.awareness, [...states.keys()]));
      send(ws, encoding.toUint8Array(a));
    }
  });
});

async function shutdown() {
  await Promise.all([...rooms.values()].map((r) => { if (r.timer) clearTimeout(r.timer); return flush(r); }));
  process.exit(0);
}
process.on('SIGINT', shutdown);
process.on('SIGTERM', shutdown);

server.listen(PORT, () => console.log(`collab instance ${INSTANCE} listening on :${PORT}`));

export default function Editor(p: Props) {

function Inner({ pageId, title, perm, user, canManage, doc, provider }: Props & { doc: Y.Doc; provider: WebsocketProvider }) {
  const [status, setStatus] = useState('connecting');
  const [synced, setSynced] = useState(false);

        ['Redo', () => editor.chain().focus().redo().run(), false],
      ]

  const live = status === 'connected';
  const tabs = ['comments', 'history', ...(canManage ? ['share'] : [])] as ('comments' | 'history' | 'share')[];

            className="w-full max-w-md border-0 bg-transparent px-0 text-3xl font-bold focus:outline-none"
            value={name}
            readOnly={!editable}
            onChange={(e) => setName(e.target.value)}

            <span className={`rounded px-2 py-0.5 text-xs ${live ? (synced ? 'bg-green-100 text-green-800' : 'bg-yellow-100 text-yellow-800') : 'bg-red-100 text-red-800'}`}>
              {live ? (synced ? 'Live' : 'Syncing…') : 'Offline – retrying'}
            </span>

type V = { id: string; label: string | null; createdAt: string; createdBy: { name: string } | null };

export function Versions({ pageId, perm, doc }: { pageId: string; perm: Perm; doc: Y.Doc }) {
  const [list, setList] = useState<V[]>([]);
  const [sel, setSel] = useState<{ id: string; snap: Y.Doc; text: string } | null>(null);
  const [err, setErr] = useState('');
  const [busy, setBusy] = useState(false);
  const base = `/api/pages/${pageId}/versions`;
  const load = useCallback(() => j(base).then(setList).catch((e) => setErr(msg(e))), [base]);
  useEffect(() => { load(); }, [load]);

  const save = async (label?: string) => { await j(base, 'POST', { label }); await load(); };
  const wrap = async (fn: () => Promise<void>) => {
    setBusy(true);
    try { await fn(); setErr(''); } catch (e) { setErr(msg(e)); }
    setBusy(false);
  };
  const open = (id: string) => wrap(async () => {
    const d = await j(`/api/versions/${id}`);
    const snap = new Y.Doc();
    Y.applyUpdate(snap, Uint8Array.from(atob(d.data), (c) => c.charCodeAt(0)));
    const text = snap.getXmlFragment('default').toString()
      .replace(/<\/(paragraph|heading|listItem|blockquote|codeBlock)>/g, '\n')
      .replace(/<[^>]+>/g, '');
    setSel({ id, snap, text });
  });
  // Restore = controlled CRDT edit of the live doc (needs EDIT; enforced by the collab server). Old state is kept as a version first.
  const restore = () => wrap(async () => {
    if (!sel || !confirm('Restore this version for all collaborators?')) return;
    await save('Before restore');
    const frag = doc.getXmlFragment('default');
    doc.transact(() => {
      frag.delete(0, frag.length);
      frag.insert(0, sel.snap.getXmlFragment('default').toArray().map((n) => n.clone() as Y.XmlElement));
    });
    setSel(null);
  });

  return (
    <div className="space-y-3 text-sm">
      {perm === 'EDIT' && <button className="btn" disabled={busy} onClick={() => wrap(() => save())}>Save version</button>}
      {err && <p className="text-red-600">{err}</p>}
      {list.length === 0 && <p className="text-gray-500">No versions yet.</p>}
      <ul className="divide-y">
        {list.map((v) => (
          <li key={v.id}>
            <button className={`w-full py-2 text-left ${sel?.id === v.id ? 'font-semibold' : ''}`} onClick={() => open(v.id)}>
              {v.label ?? 'Version'} <span className="block text-xs text-gray-500">{fmt(v.createdAt)} · {v.createdBy?.name ?? 'unknown'}</span>
            </button>
          </li>
        ))}
      </ul>
      {sel && (
        <div className="space-y-2 rounded border bg-gray-50 p-2">
          <pre className="max-h-64 overflow-auto whitespace-pre-wrap text-xs">{sel.text || '(empty)'}</pre>
          {perm === 'EDIT' && <button className="btn" disabled={busy} onClick={restore}>Restore this version</button>}
        </div>
      )}
    </div>
  );
}

export const fail = (error: string, status = 400) => NextResponse.json({ error }, { status });

export async function parse<T>(req: Request, s: z.ZodType<T, z.ZodTypeDef, unknown>): Promise<T | null> {
  const r = s.safeParse(await req.json().catch(() => null));
  return r.success ? r.data : null;
}

/** Server-side page authorization. Returns {user, perm} or {res} to return immediately. */
export async function gate(pageId: string, min: Perm, manage = false) {
  const user = await getUser();
  if (!user) return { res: fail('Unauthorized', 401) };
  const perm = await getPerm(user.id, pageId);
  if (!perm) return { res: fail('Not found', 404) };
  if (rank[perm] < rank[min] || (manage && !(await canManage(user.id, pageId)))) return { res: fail('Forbidden', 403) };
  return { user, perm };
}

// Simple in-memory sliding-window limiter (per process).
const hits = new Map<string, number[]>();
export function limited(req: Request, key: string, max = 10, windowMs = 60_000) {
  const k = key + (req.headers.get('x-forwarded-for')?.split(',')[0] ?? 'local');
  const now = Date.now();
  const a = (hits.get(k) ?? []).filter((t) => now - t < windowMs);
  a.push(now);
  hits.set(k, a);
  return a.length > max;
}

import { prisma } from './prisma';
export type Perm = 'VIEW' | 'COMMENT' | 'EDIT';
export const rank: Record<Perm, number> = { VIEW: 1, COMMENT: 2, EDIT: 3 };

/** Effective page permission. Explicit PagePermission overrides; workspace owner/member default to EDIT. */
export async function getPerm(userId: string, pageId: string): Promise<Perm | null> {
  const page = await prisma.page.findUnique({ where: { id: pageId }, select: { workspaceId: true, workspace: { select: { ownerId: true } } } });
  if (!page) return null;
  if (page.workspace.ownerId === userId) return 'EDIT';
  const pp = await prisma.pagePermission.findUnique({ where: { pageId_userId: { pageId, userId } } });
  if (pp) return pp.permission;
  const m = await prisma.workspaceMember.findUnique({ where: { workspaceId_userId: { workspaceId: page.workspaceId, userId } } });
  return m ? 'EDIT' : null;
}
export async function canManage(userId: string, pageId: string) {
  const p = await prisma.page.findUnique({ where: { id: pageId }, select: { createdBy: true, workspace: { select: { ownerId: true } } } });
  return !!p && (p.createdBy === userId || p.workspace.ownerId === userId);
}
